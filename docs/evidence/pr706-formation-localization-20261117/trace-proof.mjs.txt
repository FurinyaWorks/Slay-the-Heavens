const stringify=JSON.stringify;
JSON.stringify=function(...args){
 const input=args[0];
 if(input?.schemaVersion===1&&input?.targets){
  process.stderr.write('PROOF '+JSON.stringify({heap:process.memoryUsage().heapUsed,owners:input.owners?.length,targets:Object.keys(input.targets).length,stack:new Error().stack.split('\n').slice(2,5)})+'\n');
 }
 return Reflect.apply(stringify,this,args);
};
