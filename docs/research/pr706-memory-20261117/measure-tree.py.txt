import subprocess, resource, time, json, sys, pathlib, threading
name=sys.argv[1];command=sys.argv[2:];start=time.monotonic();peak=0;peak_count=0;done=threading.Event()
with open(name+'.log','w') as f:
 process=subprocess.Popen(command,stdout=f,stderr=subprocess.STDOUT,start_new_session=True)
 def sample():
  global peak,peak_count
  while not done.is_set():
   total=0;count=0
   for path in pathlib.Path('/proc').glob('[0-9]*/stat'):
    try:
     fields=path.read_text().split(') ',1)[1].split()
     if int(fields[2])==process.pid:
      total+=int(fields[21])*4096;count+=1
    except (OSError,ValueError,IndexError): pass
   if total>peak:peak=total;peak_count=count
   done.wait(.02)
 thread=threading.Thread(target=sample);thread.start();code=process.wait();done.set();thread.join()
usage=resource.getrusage(resource.RUSAGE_CHILDREN)
metrics=dict(command=command,exitCode=code,wallSeconds=time.monotonic()-start,maxRssKiB=usage.ru_maxrss,sampledTreeRssBytes=peak,processesAtPeak=peak_count,sampleIntervalMs=20,userSeconds=usage.ru_utime,systemSeconds=usage.ru_stime)
pathlib.Path(name+'.json').write_text(json.dumps(metrics,indent=2));print(json.dumps(metrics));sys.exit(code if code>=0 else 128-code)
